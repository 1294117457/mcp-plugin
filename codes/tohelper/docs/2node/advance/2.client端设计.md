# Node-Task 系统 - Client 端设计文档

## 一、核心概念

### 1.1 设计原则

- **复用 Tool UI 架构**：基于现有的 ToolPanel 架构扩展
- **渐进式交互**：从简单到复杂，逐步引导用户
- **可视化编排**：Task Pipeline 使用可视化编辑器
- **实时预览**：配置变更即时反馈
- **保持悬浮弹窗**：不侵入 dsh UI

### 1.2 UI 层级

```
FloatingButton (悬浮按钮)
  └─ SatelliteMenu (卫星菜单)
      ├─ 工具 → ToolPanel (已有)
      └─ 节点 → NodePanel (扩展)
          ├─ NodeList (节点列表)
          └─ NodeEditor (节点编辑器)
              ├─ BasicConfig (基础配置)
              ├─ ExecutionModeSelector (执行模式选择器)
              ├─ DirectModeConfig (Direct 模式配置)
              └─ PipelineModeConfig (Pipeline 模式配置)
                  ├─ TaskList (Task 列表)
                  └─ TaskEditor (Task 编辑器)
                      ├─ TaskTypeSelector (Task 类型选择器)
                      ├─ LLMCallConfig (LLM 调用配置)
                      ├─ ToolCallConfig (Tool 调用配置)
                      └─ TransformConfig (数据转换配置)
```

```typescript
// src/client/node/NodePanel.tsx（扩展）

import { useState, useEffect, useCallback } from 'react'
import { nodeApi, toolApi } from '../api'
import { NodeList } from './NodeList'
import { NodeEditor } from './NodeEditor'
import type { NodeConfig } from '../../types'

interface Props {
  btnPos: { x: number; y: number }
  onClose: () => void
}

const PANEL_W = 600  // 扩大宽度以容纳 Task 编辑器
const PANEL_H = 700

export function NodePanel({ btnPos, onClose }: Props) {
  const [view, setView] = useState<'list' | 'editor'>('list')
  const [editingNode, setEditingNode] = useState<NodeConfig | null>(null)
  const [nodes, setNodes] = useState<NodeConfig[]>([])
  const [equippedNodeIds, setEquippedNodeIds] = useState<string[]>([])
  const [availableTools, setAvailableTools] = useState<string[]>([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState('')

  const reload = useCallback(async () => {
    setLoading(true)
    try {
      // 并行加载 Node 列表和 Tool 列表
      const [nodeRes, toolRes] = await Promise.all([
        nodeApi.list(),
        toolApi.getTools()
      ])
      
      setNodes(nodeRes.nodes)
      setEquippedNodeIds(nodeRes.equipped)
      
      // 提取所有可用的 Tool 名称
      const toolNames = [
        ...toolRes.builtin.map(t => t.name),
        ...toolRes.mcp.map(t => t.name)
      ]
      setAvailableTools(toolNames)
      
      setLoading(false)
    } catch (e: any) {
      setError(e.message)
      setLoading(false)
    }
  }, [])

  useEffect(() => { reload() }, [reload])

  // 计算面板位置（避免超出屏幕）
  let top = btnPos.y - PANEL_H - 12
  let left = btnPos.x + 80 - PANEL_W
  if (top < 8) top = btnPos.y + 80 + 12
  if (left < 8) left = 8
  if (left + PANEL_W > window.innerWidth - 8) left = window.innerWidth - PANEL_W - 8
  if (top + PANEL_H > window.innerHeight - 8) top = window.innerHeight - PANEL_H - 8

  async function handleSave(payload: any) {
    try {
      const res = editingNode
        ? await nodeApi.update({ id: editingNode.id, ...payload })
        : await nodeApi.create(payload)
      
      if (!res.ok) {
        setError(res.error || 'Save failed')
        return
      }
      
      setView('list')
      setEditingNode(null)
      reload()
    } catch (e: any) {
      setError(e.message)
    }
  }

  async function handleDelete(id: string) {
    if (!confirm('确定删除此节点？')) return
    try {
      await nodeApi.delete(id)
      reload()
    } catch (e: any) {
      setError(e.message)
    }
  }

  async function handleEquip(nodeId: string, equipped: boolean) {
    try {
      const res = equipped
        ? await nodeApi.unequip(nodeId)
        : await nodeApi.equip(nodeId)
      
      if (!res.ok) {
        setError(res.error || 'Equip/Unequip failed')
        return
      }
      
      reload()
    } catch (e: any) {
      setError(e.message)
    }
  }

  return (
    <div className="th-panel" style={{
      position: 'fixed',
      left: `${left}px`,
      top: `${top}px`,
      width: `${PANEL_W}px`,
      height: `${PANEL_H}px`,
      pointerEvents: 'auto',
    }}>
      <div className="th-hdr">
        <h2>节点定义</h2>
        <button className="th-cls" onClick={onClose}>&times;</button>
      </div>
      
      {error && (
        <div className="th-error" style={{ padding: '8px 14px', fontSize: '13px' }}>
          {error}
          <button onClick={() => setError('')} style={{ float: 'right' }}>×</button>
        </div>
      )}
      
      <div className="th-body">
        {view === 'list' ? (
          <NodeList
            nodes={nodes}
            equippedNodeIds={equippedNodeIds}
            loading={loading}
            onEdit={(node) => {
              setEditingNode(node)
              setView('editor')
              setError('')
            }}
            onCreate={() => {
              setEditingNode(null)
              setView('editor')
              setError('')
            }}
            onDelete={handleDelete}
            onEquip={handleEquip}
          />
        ) : (
          <NodeEditor
            initial={editingNode}
            availableTools={availableTools}
            onSave={handleSave}
            onCancel={() => {
              setView('list')
              setError('')
            }}
          />
        )}
      </div>
    </div>
  )
}
```

### 3.3 NodeEditor（节点编辑器）

```typescript
// src/client/node/NodeEditor.tsx

import { useState, useEffect } from 'react'
import type { NodeConfig } from '../../types'
import { BasicConfig } from './BasicConfig'
import { ExecutionModeSelector } from './ExecutionModeSelector'
import { DirectModeConfig } from './DirectModeConfig'
import { PipelineModeConfig } from './PipelineModeConfig'

interface Props {
  initial: NodeConfig | null
  availableTools: string[]
  onSave: (payload: any) => void
  onCancel: () => void
}

export function NodeEditor({ initial, availableTools, onSave, onCancel }: Props) {
  const [step, setStep] = useState<'basic' | 'execution'>('basic')
  
  // 基础配置
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [executionMode, setExecutionMode] = useState<'direct' | 'pipeline' | 'subagent'>('direct')
  
  // Direct 模式配置
  const [systemPrompt, setSystemPrompt] = useState('')
  const [llm, setLlm] = useState({
    provider: 'deepseek-official',
    model: 'deepseek-chat',
    temperature: 0.7,
    maxTokens: 4096
  })
  
  // Pipeline 模式配置
  const [tasks, setTasks] = useState<any[]>([])
  
  // 通用配置
  const [selectedTools, setSelectedTools] = useState<string[]>([])
  const [inputSchema, setInputSchema] = useState('{\n  "type": "object",\n  "properties": {}\n}')
  const [outputSchema, setOutputSchema] = useState('{\n  "type": "object",\n  "properties": {}\n}')
  
  const [errors, setErrors] = useState<Record<string, string>>({})

  // 初始化（编辑模式）
  useEffect(() => {
    if (initial) {
      setName(initial.name)
      setDescription(initial.description)
      setExecutionMode(initial.executionMode)
      
      if (initial.executionMode === 'direct') {
        setSystemPrompt(initial.systemPrompt || '')
        if (initial.llm) setLlm(initial.llm)
      } else if (initial.executionMode === 'pipeline') {
        setTasks(initial.tasks || [])
      }
      
      setSelectedTools(initial.tools || [])
      setInputSchema(JSON.stringify(initial.inputSchema, null, 2))
      setOutputSchema(JSON.stringify(initial.outputSchema, null, 2))
    }
  }, [initial])

  function validate(): boolean {
    const newErrors: Record<string, string> = {}
    
    // 基础验证
    if (!name.trim()) newErrors.name = '节点名称不能为空'
    if (name.length < 3 || name.length > 50) newErrors.name = '节点名称长度应在 3-50 之间'
    if (!/^[a-z][a-z0-9_]*$/.test(name)) newErrors.name = '节点名称只能包含小写字母、数字和下划线，且以字母开头'
    
    if (!description.trim()) newErrors.description = '节点描述不能为空'
    if (description.length > 200) newErrors.description = '节点描述不能超过 200 字符'
    
    // 执行模式验证
    if (executionMode === 'direct') {
      if (!systemPrompt.trim()) newErrors.systemPrompt = 'System Prompt 不能为空'
    } else if (executionMode === 'pipeline') {
      if (tasks.length === 0) newErrors.tasks = '至少需要一个 Task'
    }
    
    // Schema 验证
    try {
      JSON.parse(inputSchema)
    } catch {
      newErrors.inputSchema = '输入 Schema 不是有效的 JSON'
    }
    
    try {
      JSON.parse(outputSchema)
    } catch {
      newErrors.outputSchema = '输出 Schema 不是有效的 JSON'
    }
    
    setErrors(newErrors)
    return Object.keys(newErrors).length === 0
  }

  function handleSave() {
    if (!validate()) return
    
    const payload: any = {
      name,
      description,
      executionMode,
      tools: selectedTools,
      inputSchema: JSON.parse(inputSchema),
      outputSchema: JSON.parse(outputSchema)
    }
    
    if (initial) {
      payload.id = initial.id
    }
    
    if (executionMode === 'direct') {
      payload.systemPrompt = systemPrompt
      payload.llm = llm
    } else if (executionMode === 'pipeline') {
      payload.tasks = tasks
    }
    
    onSave(payload)
  }

  return (
    <div className="th-node-editor">
      {/* 进度指示器 */}
      <div className="th-editor-steps">
        <div className={`th-step ${step === 'basic' ? 'active' : 'completed'}`}>
          1. 基础信息
        </div>
        <div className={`th-step ${step === 'execution' ? 'active' : ''}`}>
          2. 执行配置
        </div>
      </div>
      
      {/* 内容区域 */}
      <div className="th-editor-content">
        {step === 'basic' ? (
          <BasicConfig
            name={name}
            description={description}
            executionMode={executionMode}
            selectedTools={selectedTools}
            availableTools={availableTools}
            inputSchema={inputSchema}
            outputSchema={outputSchema}
            errors={errors}
            onNameChange={setName}
            onDescriptionChange={setDescription}
            onExecutionModeChange={setExecutionMode}
            onToolsChange={setSelectedTools}
            onInputSchemaChange={setInputSchema}
            onOutputSchemaChange={setOutputSchema}
          />
        ) : (
          <>
            {executionMode === 'direct' && (
              <DirectModeConfig
                systemPrompt={systemPrompt}
                llm={llm}
                errors={errors}
                onSystemPromptChange={setSystemPrompt}
                onLlmChange={setLlm}
              />
            )}
            
            {executionMode === 'pipeline' && (
              <PipelineModeConfig
                tasks={tasks}
                availableTools={selectedTools}
                errors={errors}
                onTasksChange={setTasks}
              />
            )}
            
            {executionMode === 'subagent' && (
              <div className="th-notice">
                Subagent 模式暂未实现，敬请期待
              </div>
            )}
          </>
        )}
      </div>
      
      {/* 操作按钮 */}
      <div className="th-editor-footer">
        <button className="th-btn th-btn-secondary" onClick={onCancel}>
          取消
        </button>
        
        {step === 'basic' ? (
          <button
            className="th-btn th-btn-primary"
            onClick={() => {
              if (validate()) setStep('execution')
            }}
          >
            下一步
          </button>
        ) : (
          <>
            <button
              className="th-btn th-btn-secondary"
              onClick={() => setStep('basic')}
            >
              上一步
            </button>
            <button
              className="th-btn th-btn-primary"
              onClick={handleSave}
            >
              保存
            </button>
          </>
        )}
      </div>
    </div>
  )
}
```

### 3.4 PipelineModeConfig（Pipeline 配置）

```typescript
// src/client/node/PipelineModeConfig.tsx

import { useState } from 'react'
import { TaskList } from './TaskList'
import { TaskEditor } from './TaskEditor'
import type { TaskEditorState } from '../types'

interface Props {
  tasks: TaskEditorState[]
  availableTools: string[]
  errors: Record<string, string>
  onTasksChange: (tasks: TaskEditorState[]) => void
}

export function PipelineModeConfig({ tasks, availableTools, errors, onTasksChange }: Props) {
  const [editingTaskIndex, setEditingTaskIndex] = useState<number | null>(null)

  function handleAddTask() {
    const newTask: TaskEditorState = {
      id: `task-${Date.now()}`,
      name: `Task ${tasks.length + 1}`,
      type: 'llm-call',
      llmCallConfig: {
        systemPrompt: '',
        outputFormat: 'text',
        llm: {
          provider: 'deepseek-official',
          model: 'deepseek-chat',
          temperature: 0.7,
          maxTokens: 4096,
          useNodeDefault: false
        }
      },
      expanded: true,
      errors: {}
    }
    onTasksChange([...tasks, newTask])
    setEditingTaskIndex(tasks.length)
  }

  function handleUpdateTask(index: number, updatedTask: TaskEditorState) {
    const newTasks = [...tasks]
    newTasks[index] = updatedTask
    onTasksChange(newTasks)
  }

  function handleDeleteTask(index: number) {
    const newTasks = tasks.filter((_, i) => i !== index)
    onTasksChange(newTasks)
    if (editingTaskIndex === index) {
      setEditingTaskIndex(null)
    }
  }

  function handleMoveTask(index: number, direction: 'up' | 'down') {
    if (direction === 'up' && index > 0) {
      const newTasks = [...tasks]
      ;[newTasks[index - 1], newTasks[index]] = [newTasks[index], newTasks[index - 1]]
      onTasksChange(newTasks)
    } else if (direction === 'down' && index < tasks.length - 1) {
      const newTasks = [...tasks]
      ;[newTasks[index], newTasks[index + 1]] = [newTasks[index + 1], newTasks[index]]
      onTasksChange(newTasks)
    }
  }

  return (
    <div className="th-pipeline-config">
      <div className="th-section-hdr">
        <h3>Task Pipeline</h3>
        <p className="th-hint">定义多个 Task，按顺序执行</p>
      </div>
      
      {errors.tasks && (
        <div className="th-error">{errors.tasks}</div>
      )}
      
      {editingTaskIndex === null ? (
        // 列表视图
        <TaskList
          tasks={tasks}
          onEdit={(index) => setEditingTaskIndex(index)}
          onDelete={handleDeleteTask}
          onMove={handleMoveTask}
          onAdd={handleAddTask}
        />
      ) : (
        // 编辑视图
        <TaskEditor
          task={tasks[editingTaskIndex]}
          availableTools={availableTools}
          onSave={(updatedTask) => {
            handleUpdateTask(editingTaskIndex, updatedTask)
            setEditingTaskIndex(null)
          }}
          onCancel={() => setEditingTaskIndex(null)}
        />
      )}
    </div>
  )
}
```

```typescript
        </div>
      ) : (
        <div className="th-task-items">
          {tasks.map((task, index) => (
            <div key={task.id} className="th-task-item">
              <div className="th-task-order">{index + 1}</div>
              
              <div className="th-task-content">
                <div className="th-task-hdr">
                  <h4 className="th-task-name">{task.name}</h4>
                  <span className={`th-task-badge ${task.type}`}>
                    {task.type}
                  </span>
                </div>
                
                {task.description && (
                  <p className="th-task-desc">{task.description}</p>
                )}
                
                {/* 简要配置预览 */}
                <div className="th-task-preview">
                  {task.type === 'llm-call' && task.llmCallConfig && (
                    <div className="th-task-preview-item">
                      <span className="th-label">Model:</span>
                      <span className="th-value">{task.llmCallConfig.llm.model}</span>
                    </div>
                  )}
                  
                  {task.type === 'tool-call' && task.toolCallConfig && (
                    <div className="th-task-preview-item">
                      <span className="th-label">Tool:</span>
                      <span className="th-value">{task.toolCallConfig.toolName}</span>
                    </div>
                  )}
                  
                  {task.type === 'transform' && (
                    <div className="th-task-preview-item">
                      <span className="th-label">Transform</span>
                    </div>
                  )}
                </div>
              </div>
              
              <div className="th-task-actions">
                <button
                  className="th-btn-icon"
                  title="上移"
                  disabled={index === 0}
                  onClick={() => onMove(index, 'up')}
                >
                  ↑
                </button>
                <button
                  className="th-btn-icon"
                  title="下移"
                  disabled={index === tasks.length - 1}
                  onClick={() => onMove(index, 'down')}
                >
                  ↓
                </button>
                <button
                  className="th-btn-icon"
                  title="编辑"
                  onClick={() => onEdit(index)}
                >
                  ✎
                </button>
                <button
                  className="th-btn-icon th-btn-danger"
                  title="删除"
                  onClick={() => onDelete(index)}
                >
                  ×
                </button>
              </div>
            </div>
          ))}
        </div>
      )}
      
      <button className="th-btn th-btn-primary th-btn-block" onClick={onAdd}>
        + 添加 Task
      </button>
    </div>
  )
}
```

### 3.6 TaskEditor（Task 编辑器）

```typescript
// src/client/node/TaskEditor.tsx

import { useState, useEffect } from 'react'
import type { TaskEditorState } from '../types'
import { TaskTypeSelector } from './TaskTypeSelector'
import { LLMCallConfig } from './LLMCallConfig'
import { ToolCallConfig } from './ToolCallConfig'
import { TransformConfig } from './TransformConfig'

interface Props {
  task: TaskEditorState
  availableTools: string[]
  onSave: (task: TaskEditorState) => void
  onCancel: () => void
}

export function TaskEditor({ task, availableTools, onSave, onCancel }: Props) {
  const [editedTask, setEditedTask] = useState<TaskEditorState>(task)
  const [errors, setErrors] = useState<Record<string, string>>({})

  useEffect(() => {
    setEditedTask(task)
  }, [task])

  function validate(): boolean {
    const newErrors: Record<string, string> = {}
    
    if (!editedTask.name.trim()) {
      newErrors.name = 'Task 名称不能为空'
    }
    
    if (editedTask.type === 'llm-call' && editedTask.llmCallConfig) {
      if (!editedTask.llmCallConfig.systemPrompt.trim()) {
        newErrors.systemPrompt = 'System Prompt 不能为空'
      }
    }
    
    if (editedTask.type === 'tool-call' && editedTask.toolCallConfig) {
      if (!editedTask.toolCallConfig.toolName) {
        newErrors.toolName = '请选择一个 Tool'
      }
    }
    
    if (editedTask.type === 'transform' && editedTask.transformConfig) {
      if (!editedTask.transformConfig.script.trim()) {
        newErrors.script = '转换脚本不能为空'
      }
    }
    
    setErrors(newErrors)
    return Object.keys(newErrors).length === 0
  }

  function handleSave() {
    if (!validate()) return
    onSave(editedTask)
  }

  return (
    <div className="th-task-editor">
      <div className="th-editor-hdr">
        <h3>编辑 Task</h3>
      </div>
      
      <div className="th-editor-body">
        {/* 基础信息 */}
        <div className="th-form-group">
          <label className="th-label">Task 名称 *</label>
          <input
            type="text"
            className="th-input"
            value={editedTask.name}
            onChange={(e) => setEditedTask({ ...editedTask, name: e.target.value })}
            placeholder="例如: Read File"
          />
          {errors.name && <div className="th-error-msg">{errors.name}</div>}
        </div>
        
        <div className="th-form-group">
          <label className="th-label">Task 描述</label>
          <input
            type="text"
            className="th-input"
            value={editedTask.description || ''}
            onChange={(e) => setEditedTask({ ...editedTask, description: e.target.value })}
            placeholder="可选"
          />
        </div>
        
        {/* Task 类型选择 */}
        <TaskTypeSelector
          type={editedTask.type}
          onChange={(type) => {
            // 切换类型时初始化对应配置
            const newTask: TaskEditorState = { ...editedTask, type }
            
            if (type === 'llm-call') {
              newTask.llmCallConfig = {
                systemPrompt: '',
                outputFormat: 'text',
                llm: {
                  provider: 'deepseek-official',
                  model: 'deepseek-chat',
                  useNodeDefault: false
                }
              }
              newTask.toolCallConfig = undefined
              newTask.transformConfig = undefined
            } else if (type === 'tool-call') {
              newTask.toolCallConfig = {
                toolName: '',
                inputMapping: { type: 'direct' },
                outputMapping: { type: 'direct' }
              }
              newTask.llmCallConfig = undefined
              newTask.transformConfig = undefined
            } else if (type === 'transform') {
              newTask.transformConfig = { script: '' }
              newTask.llmCallConfig = undefined
              newTask.toolCallConfig = undefined
            }
            
            setEditedTask(newTask)
          }}
        />
        
        {/* Task 类型特定配置 */}
        {editedTask.type === 'llm-call' && editedTask.llmCallConfig && (
          <LLMCallConfig
            config={editedTask.llmCallConfig}
            errors={errors}
            onChange={(config) => setEditedTask({ ...editedTask, llmCallConfig: config })}
          />
        )}
        
        {editedTask.type === 'tool-call' && editedTask.toolCallConfig && (
          <ToolCallConfig
            config={editedTask.toolCallConfig}
            availableTools={availableTools}
            errors={errors}
            onChange={(config) => setEditedTask({ ...editedTask, toolCallConfig: config })}
          />
        )}
        
        {editedTask.type === 'transform' && editedTask.transformConfig && (
          <TransformConfig
            config={editedTask.transformConfig}
            errors={errors}
            onChange={(config) => setEditedTask({ ...editedTask, transformConfig: config })}
          />
        )}
      </div>
      
      <div className="th-editor-footer">
        <button className="th-btn th-btn-secondary" onClick={onCancel}>
          取消
        </button>
        <button className="th-btn th-btn-primary" onClick={handleSave}>
          保存 Task
        </button>
      </div>
    </div>
  )
}
```

```
   - 执行模式: Pipeline
   - 选择 Tools: read_file, grep
3. 点击"下一步"
4. 添加 Task 1:
   - 名称: Read File
   - 类型: tool-call
   - Tool: read_file
   - 输入映射: Extract → {"path": "$.filepath"}
   - 输出映射: Wrap → {"filepath": "${input.filepath}", "content": "${output}"}
5. 添加 Task 2:
   - 名称: Security Analysis
   - 类型: llm-call
   - System Prompt: "You are a security expert..."
   - User Prompt Template: "File: ${input.filepath}\n\nCode:\n${input.content}"
   - Output Format: JSON
   - LLM: deepseek-chat, temperature=0.2
6. 添加 Task 3:
   - 名称: Format Report
   - 类型: transform
   - Script: "{ filepath: input.filepath, issues: input.issues, summary: `Found ${input.issues.length} issues` }"
7. 点击"保存"
8. 装配并使用
```

## 八、注意事项

### 8.1 用户体验

- **渐进式披露**：简单 Node 不显示复杂配置
- **即时反馈**：表单验证即时显示错误
- **撤销/恢复**：编辑时支持取消操作
- **自动保存**：草稿自动保存到 localStorage

### 8.2 性能优化

- **虚拟滚动**：Task 列表超过 20 个时使用虚拟滚动
- **防抖**：输入框使用防抖减少重渲染
- **懒加载**：大型 JSON Schema 编辑器按需加载

### 8.3 可访问性

- **键盘导航**：支持 Tab/Enter/Esc 操作
- **ARIA 标签**：为屏幕阅读器添加语义标签
- **焦点管理**：弹窗打开时自动聚焦到第一个输入框
